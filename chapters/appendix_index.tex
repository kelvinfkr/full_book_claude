\section*{核心概念快速定义}

\subsection*{A. 优化基础（第1--3章）}

\begin{description}
    \item[拉格朗日乘子 (Lagrange Multiplier)]
    约束优化中引入的辅助变量，表示约束放松一个单位时目标函数的改进量（"影子价格"）。
    \[
    \mathcal{L}(x, \lambda) = f(x) + \lambda g(x)
    \]

    \item[KKT条件 (Karush-Kuhn-Tucker Conditions)]
    约束优化问题最优解的必要条件，包括：
    \begin{enumerate}
        \item 驻点条件：$\nabla f + \sum \lambda_i \nabla g_i + \sum \mu_j \nabla h_j = 0$
        \item 原始可行：$g_i(x) = 0$，$h_j(x) \leq 0$
        \item 对偶可行：$\mu_j \geq 0$（不等式约束）
        \item 互补松弛：$\mu_j h_j(x) = 0$
    \end{enumerate}

    \item[对偶问题 (Dual Problem)]
    通过拉格朗日函数构造的另一个优化问题，提供原问题最优值的下界（弱对偶）。强对偶性成立时，两者最优值相等。

    \item[凸优化 (Convex Optimization)]
    目标函数和可行域都是凸的优化问题。局部最优 = 全局最优，KKT条件是充分必要条件。

    \item[泛函与Euler-Lagrange方程]
    泛函 $J[y]=\int F(x,y,y')\,dx$ 以函数为输入。极值的必要条件是
    \[
    \frac{\partial F}{\partial y} - \frac{d}{dx}\frac{\partial F}{\partial y'} = 0
    \]
    它是有限维乘子法在 $\Delta x \to 0$ 时的极限。
\end{description}

\subsection*{B. 物理与数值（第5、13章）}

\begin{description}
    \item[最小作用量原理 (Principle of Least Action)]
    真实运动使作用量 $S=\int L\,dt$ 取极值，$L=T-V$。牛顿定律是它的E-L方程。

    \item[约束力]
    带几何约束 $g(q)=0$ 的力学系统中，拉格朗日乘子 $\lambda$ 恰好等于约束力（张力、支持力）。

    \item[刚度矩阵 (Stiffness Matrix)]
    有限元把能量极小化离散为二次型 $\frac12 U^\top K U - F^\top U$，驻点条件即 $KU=F$。
\end{description}

\subsection*{C. 信息（第4章）}

\begin{description}
    \item[熵 (Entropy)]
    分布不确定性的度量，也是平均信息量：
    \[
    H(p) = -\sum_i p_i \log p_i
    \]

    \item[最大熵原理]
    在已知约束下选择熵最大的分布。乘子法给出指数族
    \[ p_i \propto \exp\Big(\sum\nolimits_j \beta_j g_j(x_i)\Big). \]

    \item[KL散度 (Kullback-Leibler Divergence)]
    两个分布之间的"差异"（非对称）：
    \[
    D_{KL}(P \| Q) = \sum_i p_i \log \frac{p_i}{q_i}
    \]

    \item[交叉熵 (Cross Entropy)]
    用分布 $Q$ 编码分布 $P$ 的平均比特数：
    \[
    H(P, Q) = -\sum_i p_i \log q_i = H(P) + D_{KL}(P \| Q)
    \]

    \item[Softmax函数]
    将实数向量转换为概率分布的函数，是最大熵分布的形式：
    \[
    p_i = \frac{e^{z_i}}{\sum_j e^{z_j}}
    \]
\end{description}

\subsection*{D. 控制与机器人（第6--7章）}

\begin{description}
    \item[雅可比转置定理]
    末端速度 $\dot x = J\dot q$，则关节力矩与末端力满足 $\tau = J^\top F$（虚功原理）。

    \item[接触互补条件]
    间隙 $\phi \ge 0$，接触力 $\lambda \ge 0$，$\lambda\phi = 0$："只有接触才有力"。

    \item[最优控制 (Optimal Control)]
    选择控制输入 $u(t)$ 使得系统轨迹最优化某个目标泛函。

    \item[伴随方程 (Adjoint Equation)]
    描述"未来代价对当前状态的敏感度"的微分方程，从终点向起点反向求解：
    \[
    \dot{\lambda} = -\frac{\partial H}{\partial x}
    \]

    \item[哈密顿函数 (Hamiltonian)]
    综合目标函数和约束的函数：
    \[
    H(x, u, \lambda) = L(x, u) + \lambda^\top f(x, u)
    \]

    \item[模型预测控制 (MPC)]
    规划多步、执行一步、用反馈修正：把优化问题嵌入反馈回路。
\end{description}

\subsection*{E. 神经网络（第8、11章，附录C）}

\begin{description}
    \item[反向传播 (Backpropagation)]
    高效计算神经网络梯度的算法，本质是伴随方程的离散版本。

    \item[计算图 (Computational Graph)]
    表示计算过程的有向无环图，前向传播计算值，反向传播计算梯度。每条边是一个等式约束。

    \item[Transformer]
    基于自注意力机制的神经网络架构，不使用循环或卷积。

    \item[注意力机制 (Attention)]
    根据Query和Key的相似度对Value加权求和：
    \[
    \text{Attention}(Q, K, V) = \text{softmax}\left(\frac{QK^\top}{\sqrt{d_k}}\right) V
    \]

    \item[FlashAttention]
    IO感知的注意力算法，通过分块计算减少内存访问。

    \item[MoE (Mixture of Experts)]
    稀疏激活的架构，每个输入只激活部分专家网络。
\end{description}

\subsection*{F. 强化学习（第9章）}

\begin{description}
    \item[贝尔曼方程 (Bellman Equation)]
    $V(s) = \max_a [r(s,a) + \gamma V(f(s,a))]$。它不是定义，而是序贯决策约束优化问题的KKT条件。

    \item[值函数 (Value Function)]
    状态转移约束的拉格朗日乘子："处在状态 $s$ 值多少分"。

    \item[对偶上升 vs 原始改进]
    Dijkstra维护最优性（对偶上升），Bellman-Ford维护可行性（原始改进）；现代RL算法都是这两种思路的变体。
\end{description}

\subsection*{G. 科学机器学习（第14--15章）}

\begin{description}
    \item[物理信息神经网络 (PINN)]
    用神经网络逼近PDE的解，把PDE残差作为约束写进损失函数（罚函数法），或用架构强制满足（硬约束）。

    \item[神经算子 (Neural Operator)]
    学习从"条件函数"到"解函数"的映射 $v(x)=\int \kappa(x,y)u(y)\,dy$，一次训练、任意条件。

    \item[格林函数 (Green's Function)]
    线性PDE的解算子的积分核，
    \[ u(x)=\int G(x,\xi) f(\xi)\,d\xi, \]
    它是解对源项的敏感度。
\end{description}

\subsection*{H. 多智能体（第10章）}

\begin{description}
    \item[纳什均衡 (Nash Equilibrium)]
    每个玩家都采用最优响应的策略组合，没有人有动机单方面偏离。所有玩家的KKT条件同时成立。

    \item[最优响应 (Best Response)]
    给定其他玩家策略时，最大化自己收益的策略。

    \item[Wardrop均衡]
    交通网络中，所有被使用的路径旅行时间相同且最小。可写成凸优化问题，其KKT条件即均衡条件。

    \item[平均场博弈 (Mean Field Game)]
    智能体数量趋于无穷时，多体博弈变成个体与密度场的交互：HJB（后向）+ FP（前向）耦合方程组。
\end{description}

\subsection*{I. 世界模型与具身智能（第16--17章）}

\begin{description}
    \item[世界模型 (World Model)]
    智能体对环境的内部表示：学习结构而非拟合像素。

    \item[控制障碍函数 (CBF)]
    安全约束 $h(x)\ge 0$ 的可微形式化：$\dot h \ge -\alpha h$。安全控制器是一个QP，其乘子是安全约束的"价格"。

    \item[可微仿真 (Differentiable Simulation)]
    物理仿真的可微分版本，允许通过反向传播（伴随方法）计算控制输入的梯度。

    \item[自由能原理 (Free Energy Principle)]
    感知、学习与行动都在最小化"惊奇"（变分自由能）的猜想。
\end{description}

    \begin{description}
    \item[视觉--语言--动作模型 (VLA)]
    以视觉--语言模型为骨干、输出动作（离散 token 或连续动作块）的机器人策略；流匹配把"生成一段动作"变成解一个 ODE。

    \item[真机强化学习 (Real-world RL)]
    直接在物理系统上学习：三条约束（动力学、安全集、数据预算）同时在场，三个乘子同时在算。
\end{description}

\subsection*{J. 智能体的后训练与数据（第12章）}

\begin{description}
    \item[RLHF 目标 (KL-regularized RL)]
    $\max_\pi \mathbb{E}_\pi[r] - \beta\,\mathrm{KL}(\pi\|\pi_{\rm ref})$：奖励是目标，"别离参考策略太远"是约束，$\beta$ 是它的价格；闭式解 $\pi^*\propto \pi_{\rm ref}\,e^{r/\beta}$ 是第4章的玻尔兹曼分布。

    \item[DPO (Direct Preference Optimization)]
    把奖励用闭式解反解并代入偏好模型，配分函数消掉，直接在策略上做回归——对偶消元。

    \item[GRPO / 可验证奖励]
    同一题采样一组回答，用组内均值作基线；奖励来自程序验证器（可行 / 不可行）。

    \item[数据流水线 (Data Pipeline)]
    生成器提议、验证器约束、过滤器执行：从约束分布采样。硬过滤对应 $\lambda\to\infty$，按 $e^{-\lambda\cdot\text{违约量}}$ 加权对应软约束。
\end{description}

\section*{本书的核心信息}

\begin{unification}[title=一个公式，多个化身]
拉格朗日函数的梯度条件
\[
\nabla_x \mathcal{L} = \nabla f + \lambda \nabla g = 0
\]
